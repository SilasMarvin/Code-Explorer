#: view $CODEVIZ_DATA/ord-ws-2.json
# Browse on this repo's own call graph (about 2,300 functions): search selects in
# place; clicks on dots and names select and move nothing; a click on empty space
# deselects; F frames the selection; the wheel zooms about the pointer and stops at
# the whole map; R shows everything from the front.
snap open
expect mode browse

key /
type spotlit
key enter
expect selection Explorer::spotlit
expect named Explorer::spotlit
expect nothing-moved
snap searched

# At the whole map the crowd round spotlit leaves most of its callers and calls
# unnamed: a name that would have to move as the selection changes goes instead.
# A click on a name or on a dot selects, and moves nothing.
click-label ui::draw_graph_labels
expect selection ui::draw_graph_labels
expect nothing-moved
snap select by name

click Explorer::spotlit
expect selection Explorer::spotlit
expect nothing-moved
snap select by dot

click Explorer::reveal_targets
expect selection Explorer::reveal_targets
expect nothing-moved
snap select its caller

click Explorer::spotlit
expect selection Explorer::spotlit
expect nothing-moved
snap select back

# Pointing names what has no room for a name, and moves nothing.
hover Explorer::spotlight
expect named Explorer::spotlight
expect nothing-moved
snap hover unnamed

key f
expect check framed
snap f

scroll 3 at Explorer::spotlit
expect check zoom-to-pointer
snap wheel in

scroll -3 at Explorer::spotlit
expect check map-on-screen
snap wheel out

click-empty
expect selection none
snap deselect

scroll -40
expect check zoom-bounds
snap wheel out far

key r
expect check framed
snap r
